"""
ICLR 2026 Outstanding Paper — "LLMs Get Lost In Multi-Turn Conversation"
Snowball mitigation: accumulate ALL requirement shards per turn.
"""

from dataclasses import dataclass, field
from typing import Optional
from openai import OpenAI   # pip install openai

@dataclass
class ConversationShard:
    turn: int
    content: str
    is_requirement: bool = True

@dataclass
class ConversationState:
    shards: list[ConversationShard] = field(default_factory=list)
    raw_history: list[dict] = field(default_factory=list)
    turn_count: int = 0

    def add_shard(self, content: str, is_requirement: bool = True):
        self.turn_count += 1
        self.shards.append(ConversationShard(
            turn=self.turn_count, content=content,
            is_requirement=is_requirement
        ))

    def build_snowball_context(self) -> str:
        """Core of the fix: concatenate ALL accumulated requirements."""
        reqs = [s for s in self.shards if s.is_requirement]
        if not reqs:
            return ""
        lines = ["[Accumulated task requirements so far:]"]
        for i, s in enumerate(reqs, 1):
            lines.append(f"  {i}. {s.content}")
        lines.append("[Please respond to ALL of the above.]")
        return "\n".join(lines)

    def reliability_score(self, responses: list[str]) -> float:
        """Proxy: response-length variance. High variance = unreliable."""
        if len(responses) < 2:
            return 1.0
        lengths = [len(r.split()) for r in responses]
        mean = sum(lengths) / len(lengths)
        variance = sum((l - mean)**2 for l in lengths) / len(lengths)
        return round(1.0 - min(variance / (mean**2 + 1e-9), 1.0), 3)


class SnowballChatAgent:
    """Drop-in wrapper: implements SNOWBALL protocol on any LLM."""

    def __init__(self, model="gpt-4o-mini",
                 system_prompt="You are a helpful assistant.",
                 api_key: Optional[str] = None, verbose=True):
        self.model, self.verbose = model, verbose
        self.system_prompt = system_prompt
        self.state = ConversationState()
        self.responses: list[str] = []
        self.client = OpenAI(api_key=api_key)

    def chat(self, user_message: str, is_requirement: bool = True) -> str:
        self.state.add_shard(user_message, is_requirement)

        # Snowball: prepend full accumulated context every turn
        ctx = self.state.build_snowball_context()
        augmented = f"{ctx}\n\nLatest: {user_message}"

        self.state.raw_history.append({"role": "user", "content": augmented})
        resp = self.client.chat.completions.create(
            model=self.model,
            messages=[{"role": "system", "content": self.system_prompt},
                      *self.state.raw_history]
        )
        reply = resp.choices[0].message.content
        self.state.raw_history.append({"role": "assistant", "content": reply})
        self.responses.append(reply)

        if self.verbose:
            r = self.state.reliability_score(self.responses)
            print(f"[Turn {self.state.turn_count}] Reliability: {r} | "
                  f"Shards: {len(self.state.shards)}")
        return reply

    def recap(self) -> str:
        """RECAP turn: give model one clean final shot at all requirements."""
        return self.chat(
            "Provide a final complete response addressing ALL requirements "
            "stated so far.", is_requirement=False
        )

The 4 failure modes

1. Premature generation — assumes missing info early, commits to wrong path
2. Verbal anchoring — over-relies on early (often wrong) answer attempts
3. Loss-of-middle-turns — ignores middle context, over-weights first + last turn
4. Verbose bloat — long responses introduce incorrect unstated assumptions

Mitigations tested in the paper

RECAP — final recapitulation turn restates all shards before final answer

SNOWBALL — every turn re-states all prior shards + new one (implemented above)

CONCAT — collapse all turns into single prompt before calling model

Key finding: performance drop is driven by unreliability, not capability loss. High-aptitude models (GPT-4.1, Gemini 2.5 Pro) are reliable in single-turn but equally unreliable in multi-turn.
